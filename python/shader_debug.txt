563:     def _compute_ascii_blend(self, t):
564:         """
565:         Returns target ascii blend 0-1 based on a periodic cycle.
566:         Waveform: ramp up, hold, ramp down, long rest.
567:         """
568:         phase = (t % self.ASCII_PERIOD) / self.ASCII_PERIOD  # 0..1
569:     ASCII_ON_FRACTION = 0.0   # DISABLED � causes abstract pattern feedback loop
570:         fade_frac = self.ASCII_FADE / self.ASCII_PERIOD
571: 
572:         # ASCII is active from phase [0.5 - on_frac/2 .. 0.5 + on_frac/2]
573:         # centred at phase 0.5
574:         start = 0.5 - on_frac / 2.0
575:         end = 0.5 + on_frac / 2.0
576: 
577:         if phase < start:
578:             return 0.0
579:         elif phase < start + fade_frac:
580:             # Fade in
581:             return smoothstep_py((phase - start) / fade_frac)
582:         elif phase < end - fade_frac:
583:             # Full ascii
584:             return 1.0
585:         elif phase < end:
586:             # Fade out
587:             return smoothstep_py(1.0 - (phase - (end - fade_frac)) / fade_frac)
588:         else:
589:             return 0.0
